"""
Real-time weather fetcher for OpenWeatherMap.

Usage:
  python real_time_weather.py all          # fetch predefined city list
  python real_time_weather.py "City,VN"    # fetch a single city

Outputs compact JSON to stdout for server.py to cache:
{
  "status": "ok",
  "updated": "2026-03-27T16:25:00Z",
  "data": [
    {"city":"Ca Mau","temp":30.4,"feels":33.1,"desc":"Có mây", ...}
  ]
}
"""

import json
import os
import sys
from datetime import datetime, timezone

import requests

# ---- CONFIG ----
API_KEY = os.getenv("OWM_API_KEY") or "7f318ae139397881686e5acd8dce296c"
# Main cities that represent the app area
DEFAULT_CITIES = [
    "Ca Mau,VN",
    "Bac Lieu,VN",
    "Nha Trang,VN",  # fallback coastal reference
]


def kmh_from_ms(speed_ms: float) -> float:
    return round(speed_ms * 3.6, 1)


def emoji_for(code: int) -> str:
    # Simple mapping sufficient for UI hero text
    if 200 <= code < 300:
        return "⛈️"
    if 300 <= code < 600:
        return "🌧️"
    if 600 <= code < 700:
        return "❄️"
    if 700 <= code < 800:
        return "🌫️"
    if code == 800:
        return "☀️"
    if 801 <= code <= 802:
        return "🌤️"
    return "☁️"


def fetch_city(city_query: str):
    url = "https://api.openweathermap.org/data/2.5/weather"
    params = {
        "q": city_query,
        "appid": API_KEY,
        "units": "metric",
        "lang": "vi",
    }
    try:
        resp = requests.get(url, params=params, timeout=10)
        resp.raise_for_status()
        data = resp.json()
        weather = data["weather"][0]
        main = data["main"]
        wind = data.get("wind", {})
        rain = data.get("rain", {})
        rain_mm = rain.get("1h") or rain.get("3h") or 0

        return {
            "city": data.get("name", city_query),
            "country": data.get("sys", {}).get("country", ""),
            "temp": round(main.get("temp", 0), 1),
            "feels": round(main.get("feels_like", main.get("temp", 0)), 1),
            "desc": weather.get("description", "").capitalize(),
            "humidity": main.get("humidity", 0),
            "wind": kmh_from_ms(wind.get("speed", 0)),
            "rain": round(rain_mm, 1),
            "icon": emoji_for(weather.get("id", 800)),
            "updated": datetime.fromtimestamp(data.get("dt", 0), tz=timezone.utc).strftime(
                "%H:%M UTC"
            ),
        }
    except Exception as e:  # noqa: BLE001 - want resilience per city
        return {"city": city_query, "error": str(e)}


def build_city_list(args):
    if len(args) >= 2 and args[1].lower() != "all":
        return [args[1]]
    return DEFAULT_CITIES


def main():
    cities = build_city_list(sys.argv)
    results = [fetch_city(c) for c in cities]
    ok = [r for r in results if "error" not in r]
    if not ok:
        # Fallback static sample so UI still has data if API/network fails.
        ok = [
            {
                "city": "Ca Mau",
                "country": "VN",
                "temp": 30.0,
                "feels": 33.0,
                "desc": "Có mây",
                "humidity": 82,
                "wind": 18.0,
                "rain": 40,
                "icon": "🌤️",
                "updated": datetime.now(timezone.utc).strftime("%H:%M UTC"),
            }
        ]
    status = "ok" if ok else "error"
    out = {
        "status": status,
        "updated": datetime.now(timezone.utc).isoformat(),
        "data": ok,
        "errors": [r for r in results if "error" in r],
    }
    print(json.dumps(out, ensure_ascii=False))


if __name__ == "__main__":
    main()
